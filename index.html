BigQuery SQL: Análise de Retenção e Ranking de Produtos (E-commerce) 
-- Consulta otimizada para Google BigQuery usando CTEs e Window Functions
WITH vendas_filtradas AS (
  SELECT 
    user_id,
    order_id,
    category,
    revenue,
    DATE(order_timestamp) AS order_date,
    ROW_NUMBER() OVER(PARTITION BY user_id ORDER BY order_timestamp ASC) AS num_pedido_cliente
  FROM `projeto-ecommerce.dw_vendas.fato_pedidos`
  WHERE DATE(order_timestamp) BETWEEN '2026-01-01' AND '2026-03-31'
    AND status = 'APPROVED'
)
SELECT 
  category,
  COUNT(DISTINCT order_id) AS total_pedidos,
  COUNT(DISTINCT CASE WHEN num_pedido_cliente = 1 THEN user_id END) AS novos_clientes,
  ROUND(SUM(revenue), 2) AS receita_total,
  ROUND(AVG(revenue), 2) AS ticket_medio
FROM vendas_filtradas
GROUP BY 1
ORDER BY receita_total DESC;

Python: Extração BigQuery + Teste A/B Estatístico para Product Discovery 
import pandas as pd
from google.cloud import bigquery
from scipy import stats

# 1. Conectar ao BigQuery e extrair dados de experimento de Produto (Discovery)
client = bigquery.Client(project="seu-projeto-gcp")

query = """
  SELECT variant, user_id, converted, session_revenue
  FROM `seu-projeto-gcp.analytics.ab_test_checkout`
  WHERE experiment_date >= DATE_SUB(CURRENT_DATE(), INTERVAL 14 DAY)
"""
df = client.query(query).to_dataframe()

# 2. Separar grupos Controle (A) vs Novo Checkout (B)
grupo_a = df[df['variant'] == 'A']['converted']
grupo_b = df[df['variant'] == 'B']['converted']

# 3. Teste T para validar significância estatística da hipótese de Produto
t_stat, p_value = stats.ttest_ind(grupo_b, grupo_a)

print(f"Taxa Conversão A: {grupo_a.mean():.2%} | Grupo B: {grupo_b.mean():.2%}")
print(f"P-valor: {p_value:.4f} -> {'Insight Válido!' if p_value < 0.05 else 'Sem diferença'}")

Apache Airflow: Estrutura de DAG para Pipeline Diário de E-commerce 
from datetime import datetime, timedelta
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.providers.google.cloud.operators.bigquery import BigQueryInsertJobOperator

default_args = {
    'owner': 'data-analytics',
    'depends_on_past': False,
    'retries': 2,
    'retry_delay': timedelta(minutes=5),
}

with DAG(
    dag_id='etl_ecommerce_kpis_diario',
    default_args=default_args,
    description='Extrai dados brutos e atualiza tabela consolidada no BigQuery',
    schedule_interval='0 6 * * *',
    start_date=datetime(2026, 1, 1),
    catchup=False,
    tags=['ecommerce', 'bigquery', 'kpis'],
) as dag:

    def validar_qualidade_dados(**kwargs):
        print("Iniciando validação de volumetria e nulos...")
        return True

    task_check = PythonOperator(
        task_id='validar_qualidade_origem',
        python_callable=validar_qualidade_dados
    )

    task_transform_bq = BigQueryInsertJobOperator(
        task_id='atualizar_tabela_gold_bigquery',
        configuration={
            "query": {
                "query": "CALL `projeto-ecommerce.dw_vendas.sp_atualiza_kpis_diarios`();",
                "useLegacySql": False,
            }
        }
    )

    task_check >> task_transform_bq

BigQuery + GA4: Funil de Conversão E-commerce usando UNNEST(event_params) 
-- Extraindo parâmetros aninhados da tabela nativa do GA4 no BigQuery
SELECT
  event_date,
  (SELECT value.string_value FROM UNNEST(event_params) WHERE key = 'page_location') AS url_pagina,
  COUNT(DISTINCT CONCAT(user_pseudo_id, 
    CAST((SELECT value.int_value FROM UNNEST(event_params) WHERE key = 'ga_session_id') AS STRING)
  )) AS sessoes_unicas,
  COUNTIF(event_name = 'view_item') AS visualizacoes_produto,
  COUNTIF(event_name = 'add_to_cart') AS adicoes_carrinho,
  COUNTIF(event_name = 'purchase') AS compras_concluidas,
  ROUND(SUM(ecommerce.purchase_revenue), 2) AS receita_ga4
FROM `projeto-ecommerce.analytics_123456789.events_*`
WHERE _TABLE_SUFFIX BETWEEN '20260301' AND '20260331'
GROUP BY 1, 2
HAVING visualizacoes_produto > 50
ORDER BY receita_ga4 DESC;
